% 2022-bebb-gk-B2.2a-m1 (Original 2022-bebb-gk-B2.2a, Schwere I, rang 1; herausgelöste Fassung: keine)
\begin{pfaufg}{GK~’22\\B2.2a}{1.}{}
Gegeben ist die Funktion $f$ mit $f(x) = -\tfrac{1}{3}x^3 + x^2$, $x \in \mathbb{R}$. Gib das Verhalten der Funktionswerte von $f$ für $x \to +\infty$ und für $x \to -\infty$ an.
\karo{3}
\fusshilfe{\mbox{1:~für x $\to$ +∞: f(x) $\to$ $-$∞; für x $\to$ $-$∞: f(x) $\to$ +∞}}
\end{pfaufg}
% 2026-bb-gk-B2.1a-m1 (Original 2026-bb-gk-B2.1a, Schwere I, rang 2; herausgelöste Fassung: keine)
\begin{pfaufg}{GK~’26\\B2.1a}{2.}{}
Gegeben ist die in $\mathbb{R}$ definierte Funktion $f$ mit $f(x) = \tfrac{2}{3}x^4 + \tfrac{4}{3}x^3 + 2$ und ihrer Ableitung $f'(x) = \tfrac{8}{3}x^3 + 4x^2$. Gib das Verhalten von $f'$ für $x \to -\infty$ und für $x \to +\infty$ an.
\karo{3}
\fusshilfe{\mbox{2:~für x $\to$ $-$∞: f$'$ $\to$ $-$∞; für x $\to$ +∞: f$'$ $\to$ +∞}}
\end{pfaufg}

% 2022-bebb-gk-B2.1b-m1 (Original 2022-bebb-gk-B2.1b, Schwere I, rang 3; herausgelöste Fassung: keine)
\begin{pfaufg}{GK~’22\\B2.1b}{3.}{}
Gegeben ist die in $\mathbb{R}$ definierte Funktion $f$ mit $f(x) = (x + 3) \cdot e^{-x}$ und ihrer Ableitung $f'(x) = -(x + 2) \cdot e^{-x}$. Gib den Grenzwert von $f$ für $x \to +\infty$ an. Beschreibe, wie der Graph von $f$ für große $x$ verläuft.
\karo{3}
\fusshilfe{\mbox{3:~Grenzwert 0; der Graph nähert sich für große x der x-Achse von oben}}
\end{pfaufg}
% 2023-bebb-gk-B2.1a-m1 (Original 2023-bebb-gk-B2.1a, Schwere I, rang 4; herausgelöste Fassung: keine)
\begin{pfaufg}{GK~’23\\B2.1a}{4.}{}
Gegeben ist die in $\mathbb{R}$ definierte Funktion $f$ mit $f(x) = 0{,}5 \cdot (x^2 - 9) \cdot e^x$; ihr Graph ist $G$. Gib das Verhalten der Funktionswerte von $f$ für $x \to +\infty$ und für $x \to -\infty$ an.
\karo{3}
\fusshilfe{\mbox{4:~f(x) $\to$ +∞; f(x) $\to$ 0}}
\end{pfaufg}
% 2025-bebb-gk-B2.2a-m1 (Original 2025-bebb-gk-B2.2a, Schwere I, rang 5; herausgelöste Fassung: keine)

\begin{pfaufg}{GK~’25\\B2.2a}{5.}{}
Gegeben ist die in $\mathbb{R}$ definierte Funktion $f$ mit $f(x) = (3 - x) \cdot e^x$; die Abbildung zeigt ihren Graphen. Gib das Verhalten von $f$ für $x \to -\infty$ und für $x \to +\infty$ an.
\gzwei{\begin{abiksys}[xmin=-5,xmax=3.5,ymin=-1,ymax=8,le=0.8,yle=0.4,ystep=1]\funktion[1]{(3-\x)*exp(\x)}{G_f}\end{abiksys}}{\karo{3}}
\fusshilfe{\mbox{5:~f(x) $\to$ 0; f(x) $\to$ $-$∞}}
\end{pfaufg}
