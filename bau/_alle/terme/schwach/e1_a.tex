\begin{aufgabe}{Term aufstellen}
\begin{teile}
\teil Das Vierfache einer Zahl $x$, um 1 vermindert – welcher Term passt? \\ \kreuz{$4x - 1$} \\ \kreuz{$4 \cdot (x - 1)$} \\ \kreuz{$1 - 4x$}
\end{teile}
\begin{teile}
\teil Eine Zahl $x$ wird verdoppelt und um 9 vergrößert – welcher Term passt? \\ \kreuz{$2 \cdot (x + 9)$} \\ \kreuz{$2x + 9$} \\ \kreuz{$9x + 2$}
\end{teile}
\begin{teile}
\teil Die Hälfte einer Zahl $x$, um 5 vergrößert – welcher Term passt? \\ \kreuz{$(x + 5) : 2$} \\ \kreuz{$2x + 5$} \\ \kreuz{$x : 2 + 5$}
\end{teile}
\begin{teile}
\teil Das Dreifache einer Zahl $x$, um 10 vergrößert – welcher Term passt? \\ \kreuz{$3x + 10$} \\ \kreuz{$10x + 3$} \\ \kreuz{$3 \cdot (x + 10)$}
\end{teile}
\begin{teile}
\teil Das Fünffache einer Zahl $x$, um 4 vergrößert – welcher Term passt? \\ \kreuz{$4x + 5$} \\ \kreuz{$5 \cdot (x + 4)$} \\ \kreuz{$5x + 4$}
\end{teile}
