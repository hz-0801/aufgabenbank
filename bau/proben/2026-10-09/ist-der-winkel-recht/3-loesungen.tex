\begin{pfloesung}{}

\begin{pfloesung}{}

\lz{b)}{nein}{$60^2 + 90^2 = 11\,700 \ne 12\,100 = 110^2$}{}
\lz{c)}{ja, rechter Winkel bei $A$}{$20^2 + 21^2 = 841 = 29^2$; $A$ liegt gegenüber $\overline{BC}$}{}
\end{pfloesung}
\begin{pfloesung}{}
\lz{5a)}{$3{,}7$ dm}{Hypotenuse gesucht: $1{,}44 + 12{,}25 = 13{,}69$ \ $\Rightarrow$ \ $\sqrt{13{,}69} = 3{,}7$}{}
\lz{b)}{$2{,}1$ m}{Kathete gesucht: $8{,}41 - 4 = 4{,}41$ \ $\Rightarrow$ \ $\sqrt{4{,}41} = 2{,}1$}{}
\lz{c)}{$9$ cm}{Kathete gesucht, in cm: $41^2 - 40^2 = 81$ \ $\Rightarrow$ \ $\sqrt{81} = 9$}{}
\end{pfloesung}
\begin{pfloesung}{}

\begin{pfloesung}{}

\end{pfloesung}

\end{pfloesung}
