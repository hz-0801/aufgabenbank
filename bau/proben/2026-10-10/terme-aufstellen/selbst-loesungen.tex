\noindent{\Large\bfseries Lösungen}\hspace{1em}{\color{mbgrau}Terme aufstellen}\par\smallskip
\noindent{\small Links das Ergebnis, rechts der Weg in kurzen Schritten. Stimmt dein Ergebnis nicht, such den ersten Schritt, der anders ist.}\par
\par\Needspace{25mm}\vspace{6pt}\noindent\colorbox{black!12}{\makebox[7mm]{\bfseries\strut A}}\hspace{2mm}{\bfseries Rechenwörter in Terme übersetzen}\par\vspace{4pt}
\lsg{1.}{$5x$;\ $x + 8$;\ $x - 6$;\ $x : 2$;\ $10 - x$;\ $7x$}{a) $5x$; b) $x + 8$; c) $x - 6$; d) $x : 2$; e) $10 - x$; f) $7x$}
\lsg{2.}{$7x - 3$}{erst das Siebenfache $7x$, dann minus 3.}
\lsg{3.}{$3x + 4$;\ $4x - 9$;\ $20 - 2x$;\ $x : 2 + 1$}{a) $3x + 4$; b) $4x - 9$; c) $20 - 2x$; d) $x : 2 + 1$}
\lsg{4.}{a) $x + 4$; \ $3x$ \quad b) $36$ Jahre, ja}{a) Tim: $x + 4$; Mutter: $3x$. b) Tim: $12 + 4 = 16$; Mutter: $3 \cdot 12 = 36$. $36 > 35$: ja, die Mutter ist älter als 35.}
\par\Needspace{25mm}\vspace{6pt}\noindent\colorbox{black!12}{\makebox[7mm]{\bfseries\strut B}}\hspace{2mm}{\bfseries Klammer, wenn ein Ergebnis weiterverarbeitet wird}\par\vspace{4pt}
\lsg{1.}{$3 \cdot (x + 5)$;\ $4 \cdot (x - 2)$;\ $(x + 8) : 2$}{a) $3 \cdot (x + 5)$; b) $4 \cdot (x - 2)$; c) $(x + 8) : 2$}
\lsg{2.}{$3 \cdot (2x + 5)$;\ $(3x + 4) : 2$;\ $4x + 6$}{a) $3 \cdot (2x + 5)$; b) $(3x + 4) : 2$; c) $4x + 6$ (ohne Klammer: nach dem Addieren geschieht nichts mehr)}
\lsg{3.}{$3 \cdot (5x - 4)$}{Fünffaches $5x$, Differenz $5x - 4$, als Ganzes verdreifacht.}
\lsg{4.}{$3 \cdot (x + 5)$; \ $42$ Euro, nein}{jeder zahlt $x + 5$, drei Personen. Dann $3 \cdot (9 + 5) = 3 \cdot 14 = 42$ Euro; $42 > 40$: nein, es fehlen 2 Euro.}
\par\Needspace{25mm}\vspace{6pt}\noindent\colorbox{black!12}{\makebox[7mm]{\bfseries\strut C}}\hspace{2mm}{\bfseries Term zur Figur: Umfang und Flächeninhalt}\par\vspace{4pt}
\lsg{1.}{$4a$;\ $2x + 10$ und $5x$;\ $3s$}{a) $u = 4 \cdot a$; b) $u = 2x + 10$ (auch $x + 5 + x + 5$), $A = 5 \cdot x$; c) $u = 3 \cdot s$}
\lsg{2.}{$a \cdot (b + c)$}{die ganze Höhe ist $b + c$, mal die Breite $a$.}
\lsg{3.}{a) $y \cdot (6 + 3)$ und $6y + 3y$ \quad b) $2y + 18$}{a) $A = y \cdot (6 + 3) = 9y$ \ oder \ $A = 6 \cdot y + 3 \cdot y$. b) $u = 6 + 3 + y + 6 + 3 + y = 2y + 18$}
\lsg{4.}{a) $2x + 6$ \quad b) $18$\,m, ja \quad c) $18$\,m²}{a) Skizze: Rechteck mit den Seiten $x$ und 3; Zaun $= x + 3 + x + 3 = 2x + 6$. b) $2 \cdot 6 + 6 = 18$\,m; $18 < 20$: ja, 2\,m bleiben übrig. c) $A = 3 \cdot 6 = 18$\,m².}
\par\Needspace{25mm}\vspace{6pt}\noindent\colorbox{black!12}{\makebox[7mm]{\bfseries\strut D}}\hspace{2mm}{\bfseries Terme in Sachaufgaben}\par\vspace{4pt}
\lsg{1.}{$4 + 5n$; \ $19$ Euro}{$4 + 5 \cdot n$; \ $4 + 5 \cdot 3 = 19$ Euro}
\lsg{2.}{a) $4a + 2b$ \quad b) $10$ Euro}{a) $4a + 2b$. b) $4 \cdot 1{,}50 + 2 \cdot 2 = 6 + 4 = 10$ Euro}
\lsg{3.}{a) fest 2 Euro, je Stunde 1,50 Euro, $x$ Stunden \quad b) $8$ Euro}{a) 2: fester Betrag (Einfahrt); 1,50: Preis je Stunde; $x$: Zahl der Stunden. b) $2 + 1{,}50 \cdot 4 = 2 + 6 = 8$ Euro}
\lsg{4.}{Fluss ($22$ Euro statt $24$ Euro), um $2$ Euro billiger}{Fluss: $10 + 2 \cdot h$; See: $4 \cdot h$ ($h$: Stunden). Für 6 Stunden: Fluss $10 + 2 \cdot 6 = 22$ Euro; See $4 \cdot 6 = 24$ Euro. Fluss ist billiger, um $24 - 22 = 2$ Euro.}
\par\Needspace{25mm}\vspace{6pt}\noindent\colorbox{black!12}{\makebox[7mm]{\bfseries\strut T}}\hspace{2mm}{\bfseries Probetest}\par\vspace{4pt}
\lsg{1.}{$8x - 6$}{Achtfaches $8x$, dann minus 6.}
\lsg{2.}{$(2x + 7) : 2$}{Doppeltes $2x$, Summe $2x + 7$, als Ganzes halbiert.}
\lsg{3.}{$A = 5 \cdot (x + 2)$; \ $u = 2x + 14$}{$A = 5 \cdot (x + 2)$ oder $5x + 10$; \ $u = 2x + 14$}
\lsg{4.}{$6 \cdot (x - 3)$}{\ (auch $(x - 3) \cdot 6$)}
\lsg{5.}{a) $5 \cdot (s + t)$ \quad b) $350$}{a) $5 \cdot (s + t)$ (auch $5s + 5t$). b) $5 \cdot (40 + 30) = 5 \cdot 70 = 350$ Fahrgäste}
\lsg{6.}{Fit $20 + 15m$, Top $25m$, ab dem 3. Monat Fit}{a) Fit: $20 + 15 \cdot m$; Top: $25 \cdot m$. b) Fit: 35; 50; 65; 80; 95. Top: 25; 50; 75; 100; 125. c) Nach 2 Monaten kosten beide 50 Euro; ab dem 3. Monat ist Fit günstiger.\par\smallskip \adjustbox{max width=\linewidth,max height=4cm}{\begin{varwidth}{50cm}\wertetabelle[35,50,65,80,95]{m}{\text{Fit}}{1,2,3,4,5}\par\wertetabelle[25,50,75,100,125]{m}{\text{Top}}{1,2,3,4,5}\end{varwidth}}}
