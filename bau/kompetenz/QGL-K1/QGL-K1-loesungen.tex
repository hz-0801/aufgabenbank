\kbloesung{2.}{$x^2 + 4x$}
\kbloesung{3.}{$7$}

\kbloesung{6.}{$x_{1,2} = 2 \pm \sqrt{4 + 12} = 2 \pm 4$; $x_1 = 6$, $x_2 = -2$}
\kbloesung{7.}{$x^2 - 6x + 5 = 0$; $x_{1,2} = 3 \pm \sqrt{9 - 5} = 3 \pm 2$; $x_1 = 5$, $x_2 = 1$}
\kbloesung{8.}{durch $5$ geteilt: $x^2 - 4x + 3 = 0$; $x_{1,2} = 2 \pm \sqrt{4 - 3} = 2 \pm 1$; $x_1 = 3$, $x_2 = 1$}

\kbloesung{10.}{$x_{1,2} = 5 \pm \sqrt{25 - 25} = 5$ – genau eine Lösung: $x = 5$}
\kbloesung{11.}{Unter der Wurzel: $1 - 5 = -4$, negativ: keine Lösung}
\kbloesung{12.}{$D = 16 - 7 = 9$, positiv: zwei Lösungen}
\kbloesung{13.}{$x_{1,2} = 3 \pm \sqrt{9 - 4} = 3 \pm \sqrt{5}$; $x_1 \approx 5{,}24$, $x_2 \approx 0{,}76$}
\kbloesung{14.}{$(-2)^2 - 3 \cdot (-2) - 10 = 4 + 6 - 10 = 0$ (wA): ja}
\kbloesung{15.}{$x^2 + 4x + 4 = 5x + 10$; $x^2 - x - 6 = 0$; $x_{1,2} = 0{,}5 \pm \sqrt{0{,}25 + 6} = 0{,}5 \pm 2{,}5$; $x_1 = 3$, $x_2 = -2$}
\kbloesung{16.}{Nullprodukt: $x \cdot (x - 12) = 0$; $x_1 = 0$, $x_2 = 12$}
\kbloesung{17a)}{$x_{1,2} = 5 \pm \sqrt{25 - 22} = 5 \pm \sqrt{3}$; $x_1 \approx 6{,}73$, $x_2 \approx 3{,}27$; Schnittpunkte $(6{,}73|0)$, $(3{,}27|0)$}
\kbloesung{17b)}{$x^2 - 2 = 3x + 2$; $x^2 - 3x - 4 = 0$; $x_{1,2} = 1{,}5 \pm \sqrt{2{,}25 + 4} = 1{,}5 \pm 2{,}5$; $x_1 = 4$, $x_2 = -1$; Schnittpunkte $(4|14)$ und $(-1|-1)$}
\kbloesung{17c)}{$-x^2 + 2x + 6 = -2$; $-x^2 + 2x + 8 = 0 \mid : (-1)$; $x^2 - 2x - 8 = 0$; $x_{1,2} = 1 \pm \sqrt{1 + 8} = 1 \pm 3$; $x_1 = 4$, $x_2 = -2$}
\kbloesung{17d)}{$x^2 - 6x + 9 - 2 = -2x + 19$; $x^2 - 4x - 12 = 0$; $x_{1,2} = 2 \pm \sqrt{4 + 12} = 2 \pm 4$; $x_1 = 6$, $x_2 = -2$; Schnittpunkte $(6|7)$ und $(-2|23)$}
